# Extract data from each card
for page in range(1, 16):

    if page == 1:
        url = "https://internshala.com/internships/"
    else:
        url = "https://internshala.com/internships/page-{page}/"

    print("Scraping page:", page)

    driver.get(url)
    time.sleep(3)

    soup = BeautifulSoup(driver.page_source, "html.parser")

    cards = soup.find_all("div", class_="individual_internship")

    print("Internships found:", len(cards))

    for card in cards:

        # Internship Title
        title = card.find("a", class_="job-title-href")

        # Company
        company = card.find("p", class_="company-name")

        # Stipend
        stipend = card.find("span", class_="stipend")

        # Location
        location = card.find("div", class_="locations")

        # Duration
        duration = card.find(
            "span",
            string=lambda text: text and "Months" in text
        )

        # Skills
        skills = card.find_all("div", class_="job_skill")

        # Posted
        posted = card.find("div", class_="status-success")

        # Convert to text
        title = title.get_text(strip=True) if title else None

        company = company.get_text(strip=True) if company else None

        stipend = stipend.get_text(strip=True) if stipend else None

        location = location.get_text(" ", strip=True) if location else None

        duration = duration.get_text(strip=True) if duration else None

        skills = ", ".join(
            skill.get_text(strip=True)
            for skill in skills
        )

        posted = posted.get_text(strip=True) if posted else None

        # Store data
        data.append({
            "Internship_Title": title,
            "Company": company,
            "Location": location,
            "Stipend": stipend,
            "Duration": duration,
            "Skills": skills,
            "Posted": posted
        })

print("Scraping completed!")
print("Total records:", len(data))

def two_tail_t_test(population_mean, sample_size,
                    sample_mean, sample_std, cl):

    alpha = 1 - cl
    area = 1 - alpha / 2

    t_critical = st.t.ppf(area, df=sample_size - 1)

    t_stats = (sample_mean - population_mean) / (
        sample_std / (sample_size ** 0.5)
    )

    print("T-critical value:", t_critical)
    print("T-statistic:", t_stats)

    if np.abs(t_stats) > t_critical:
        print("Reject the null hypothesis")
    else:
        print("Fail to reject the null hypothesis")
